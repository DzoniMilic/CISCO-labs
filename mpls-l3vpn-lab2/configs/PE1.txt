! Device: PE1
! Role: Provider Edge - Customer Site A
! Platform: Cisco IOS 15.6
!
! Customer VRF
ip vrf CUSTOMER_A
rd 65000:10
route-target export 65000:10
route-target import 65000:10
!
! PE loopback used for routing and MP-BGP
interface Loopback1
ip address 11.11.11.11 255.255.255.255
!
! Provider core link to P1
interface GigabitEthernet0/1
ip address 10.10.10.1 255.255.255.252
mpls ip
!
! Customer-facing interface toward CE1
interface GigabitEthernet0/2
ip vrf forwarding CUSTOMER_A
ip address 192.168.10.2 255.255.255.0
!
! Provider OSPF
router ospf 1
mpls ldp autoconfig
network 10.10.10.0 0.0.0.3 area 0
network 11.11.11.11 0.0.0.0 area 0
!
! MP-BGP VPNv4
router bgp 65000
bgp log-neighbor-changes
neighbor 44.44.44.44 remote-as 65000
neighbor 44.44.44.44 update-source Loopback1
!
address-family vpnv4
neighbor 44.44.44.44 activate
neighbor 44.44.44.44 send-community extended
exit-address-family
!
address-family ipv4 vrf CUSTOMER_A
redistribute connected
exit-address-family
!
! LDP router ID
mpls ldp router-id Loopback1 force
